<script>
    // Parallax-ish effect for scattered layers
    window.addEventListener('scroll', () => {
        const scrolled = window.scrollY;
        const layers = document.querySelectorAll('.scatter-1, .scatter-2, .scatter-3, .scatter-4');
        
        layers.forEach((layer, index) => {
            const speed = (index + 1) * 0.05;
            const yPos = -(scrolled * speed);
            // Preserving original transform and adding scroll offset
            const currentTransform = getComputedStyle(layer).transform;
            layer.style.transform = `${currentTransform === 'none' ? '' : currentTransform} translateY(${yPos}px)`;
        });

        const header = document.querySelector('header');
        if (scrolled > 50) {
            header.classList.add('bg-white/80', 'backdrop-blur-md', 'shadow-sm', 'border-b', 'border-black/5');
        } else {
            header.classList.remove('bg-white/80', 'backdrop-blur-md', 'shadow-sm', 'border-b', 'border-black/5');
        }
    });

    // Smooth scroll
    document.querySelectorAll('a[href^="#"]').forEach(anchor => {
        anchor.addEventListener('click', function (e) {
            e.preventDefault();
            const target = document.querySelector(this.getAttribute('href'));
            if (target) {
                target.scrollIntoView({ behavior: 'smooth' });
            }
        });
    });

    // Interaction Observer for revealing sections on scroll
    const observerOptions = {
        threshold: 0.1
    };

    const revealObserver = new IntersectionObserver((entries) => {
        entries.forEach(entry => {
            if (entry.isIntersecting) {
                entry.target.classList.add('animate-reveal');
                revealObserver.unobserve(entry.target);
            }
        });
    }, observerOptions);

    // Any items that aren't revealed via standard timing (like late features)
    document.querySelectorAll('.kinetic-layer > div, .kinetic-layer > section').forEach(el => {
        if (!el.classList.contains('animate-reveal')) {
            revealObserver.observe(el);
        }
    });
</script>
